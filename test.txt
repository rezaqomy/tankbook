from src.shared_kernel.repository.async_abstract_repository import AsyncRepository
from src.auth.adapters.data_models.user_data_model import User as UserDB
from src.auth.domain.entities.user import User as UserEntity
from sqlalchemy.ext.asyncio import AsyncSession
from src.auth.adapters.mappers import UserMapper
from src.shared_kernel.config import settings

JWT_EXP = settings.JWT_EXP


class AsyncUserRepository(AsyncRepository[UserDB, int]):
    def __init__(self, session: AsyncSession) -> None:
        self._session = session

    async def create(self, user: UserEntity) -> UserEntity:
        user.update_expiration(JWT_EXP)
        model_user = UserMapper.entity_to_model(user)
        self.session.add(model_user)
        await self.session.flush()
        user = UserMapper.model_to_entity(model_user)
        return user

    async def find_by_username(self, username: str) -> UserEntity:
        user = await self.find_by_col(username=username)
        return UserMapper.model_to_entity(user)

    async def get_by_id(self, id: int) -> UserEntity:
        user = await self.find_by_col(id=id)
        return UserMapper.model_to_entity(user)

    async def update_expiration_date(self, user: UserEntity):
        user.update_expiration(JWT_EXP)
        model_user = UserMapper.entity_to_model(user)
        await self.session.flush(model_user)
        return UserMapper.model_to_entity(model_user)

import re
from typing import Optional
from pydantic import BaseModel, Field, field_validator

from src.shared_kernel.config import EMAIL_REGEX, IRAN_PHONE_NUMBER_REGEX


def clean_phone_number(phone_number: str) -> str:
    phone_number = re.sub(r"\D", "", phone_number)

    if phone_number.startswith("98"):
        phone_number = "0" + phone_number[2:]

    if len(phone_number) != 11:
        raise ValueError("Phone number must be 11 digits.")

    return phone_number


class BaseUserCommand(BaseModel):
    username: str = Field(nullable=False)


class RegisterUserCommand(BaseUserCommand):
    first_name: Optional[str] = Field(default=None)
    last_name: Optional[str] = Field(default=None)
    email: Optional[str] = Field(default=None, pattern=EMAIL_REGEX)
    phone_number: Optional[str] = Field(default=None, pattern=IRAN_PHONE_NUMBER_REGEX)
    password: str

    @field_validator("phone_number")
    def validate_phone_number(cls, v):
        return clean_phone_number(v)

from dataclasses import dataclass
from typing import Optional
from datetime import datetime, timedelta, timezone
from jose import jwt
import bcrypt


from src.auth.domain.commands import RegisterUserCommand
from src.auth.entrypoints.query_params import Token
from src.shared_kernel.domain.abstract_entity import Entity
from src.auth.enums import UserRoles
from src.shared_kernel.config import settings

JWT_SECRET = settings.JWT_SECRET
JWT_ALG = settings.JWT_ALG


@dataclass
class User(Entity):
    id: Optional[int] = None
    username: Optional[str] = None
    first_name: Optional[str] = None
    last_name: Optional[str] = None
    email: Optional[str] = None
    phone_number: Optional[str] = None
    password: Optional[str] = None
    role: UserRoles = UserRoles.CUSTOMER
    expiration_token_date: Optional[datetime] = None

    def set_password(self, password: str) -> None:
        if not password:
            raise ValueError("Password cannot be empty")
        self.password = self.hash_password(password)

    def hash_password(self, password: str) -> str:
        pw = bytes(password, "utf-8")
        salt = bcrypt.gensalt()
        return bcrypt.hashpw(pw, salt).decode("utf-8")

    def verify_password(self, password: str) -> bool:
        if not password or not self.password:
            return False
        return bcrypt.checkpw(password.encode("utf-8"), self.password.encode("utf-8"))

    def update_expiration(self, value_by_seconds: int) -> None:
        if value_by_seconds < 0:
            raise ValueError("Expiration time cannot be negative")
        self.expiration_token_date = datetime.now(timezone.utc) + timedelta(
            seconds=value_by_seconds
        )

    @property
    def access(self) -> str:
        if not self.expiration_token_date:
            raise ValueError("Expiration date is not set.")
        exp = self.expiration_token_date.timestamp()
        data = {
            "exp": exp,
            "id": self.id,
        }
        return jwt.encode(data, JWT_SECRET, algorithm=JWT_ALG)

    @property
    def token(self) -> Token:
        return Token(access=self.access)

    @staticmethod
    def register_user(command: RegisterUserCommand) -> "User":
        return User(**command.dict())

    def __repr__(self):
        return f"<User {self.username}>"

class AsyncBookRepository(AsyncRepository[BookDB, int]):
    def __init__(self, session: AsyncSession) -> None:
        self._session = session
        self.gener = AsyncGenerRepository(session)

    async def create(self, instance: BookEntity) -> BookEntity:
        existing_book = await self.find_by_col(isbn=instance.isbn)
        if existing_book:
            raise ISBNAlreadyExistsException()

        gener = await self.gener.get_by_id(instance.gener)
        if not gener:
            raise GenerNotFound()

        book_model = BookMapper.entity_to_model(instance)
        self.session.add(book_model)
        await self.session.flush()
        book = BookMapper.model_to_entity(instance=book_model)
        return book

    async def get_by_id(self, id: int) -> BookEntity:
        book_model = await self.find_by_col(id=id)
        if not book_model:
            raise BookNotFound
        gener = await self.gener.get_by_id(book_model.gener)
        return BookMapper.model_to_entity(instance=book_model)

    async def get_all_by_pagination(
        self, page: int, per_page: int
    ) -> tuple[List[BookEntity], int]:
        books_model, count = await self.paginate(page=page, per_page=per_page)
        books = []
        for book in books_model:
            gener = await self.gener.get_by_id(book.gener)
            book_entity = BookMapper.model_to_entity(instance=book)
            books.append(book_entity)
        return books, count

    async def update(self, id: int, book_data: dict) -> BookEntity:
        book_model = await self.find_by_col(id=id)
        if not book_model:
            raise BookNotFound
        if "isbn" in book_data and book_model.isbn != book_data["isbn"]:
            print(book_data)
            existing_book = await self.find_by_col(isbn=book_data["isbn"])
            if existing_book:
                raise ISBNAlreadyExistsException()
        if "gener" in book_data and book_model.gener != book_data["gener"]:
            gener = await self.gener.get_by_id(id=book_data["gener"])
            if not gener:
                raise GenerNotFound

        for key, value in book_data.items():
            setattr(book_model, key, value)

        await self.session.flush()
        gener = await self.gener.get_by_id(book_model.gener)
        return BookMapper.model_to_entity(instance=book_model)

    async def delete_by_id(self, id: int):
        book_model = await self.find_by_col(id=id)
        if not book_model:
            raise BookNotFound
        await self.delete(book_model)

===== src/notification/circuit_breaker.py =====
class CircuitBreaker:
    def __init__(self, providers):
        self.providers = providers
        self.failed_providers = set()

    def send_message(self, phone_number: str, message: str) -> bool:
        for provider in self.providers:
            if provider in self.failed_providers:
                continue
            if provider.send_message(phone_number, message):
                return True
            else:
                self.failed_providers.add(provider)
        
        # Reset failed providers after trying all
        self.failed_providers.clear()
        return False

import random
from datetime import datetime, timedelta
from src.notification.circuit_breaker import CircuitBreaker
from src.shared_kernel.redis_cache import RedisCache
from src.notification.sms_providers.kave_negari_provider import KaveNegariProvider
from src.notification.sms_providers.signal_provider import SignalProvider

class OTPService:
    def __init__(self):
        self.cache = RedisCache()
        self.providers = [
            KaveNegariProvider(),
            SignalProvider(),
            SMSIRProvider()
        ]
        self.circuit_breaker = CircuitBreaker(self.providers)

    def generate_otp(self) -> str:
        return str(random.randint(100000, 999999))

    def send_otp(self, phone_number: str) -> bool:
        if not self.check_rate_limit(phone_number):
            return False

        otp = self.generate_otp()
        if self.circuit_breaker.send_message(phone_number, otp):
            self.cache.set(phone_number, otp)
            self.increment_rate_limit(phone_number)
            return True
        return False

    def validate_otp(self, phone_number: str, otp: str) -> bool:
        cached_otp = self.cache.get(phone_number)
        if cached_otp and cached_otp == otp:
            self.cache.delete(phone_number)
            self.reset_rate_limit(phone_number)
            return True
        return False

    def check_rate_limit(self, phone_number: str) -> bool:
        short_term_limit = self.cache.get(f"{phone_number}_short_term")
        long_term_limit = self.cache.get(f"{phone_number}_long_term")

        short_term_limit = int(short_term_limit) if short_term_limit else 0
        long_term_limit = int(long_term_limit) if long_term_limit else 0

        if short_term_limit >= 5 or long_term_limit >= 10:
            return False
        return True

    def increment_rate_limit(self, phone_number: str):
        short_term_key = f"{phone_number}_short_term"
        long_term_key = f"{phone_number}_long_term"

        self.cache.incr(short_term_key)
        self.cache.incr(long_term_key)

        self.cache.expire(short_term_key, 120)  # 2 minutes
        self.cache.expire(long_term_key, 3600)  # 1 hour

    def reset_rate_limit(self, phone_number: str):
        self.cache.delete(f"{phone_number}_short_term")
        self.cache.delete(f"{phone_number}_long_term")

from typing import List
from sqlalchemy.ext.asyncio import AsyncSession


from src.profiles.adapters.mapper import CustomerMapper
from src.shared_kernel.exceptions import CustomerNotFound, UserNotFound
from src.shared_kernel.repository.async_abstract_repository import AsyncRepository

from src.profiles.domain.entities.customer import Customer as CustomerEntity
from src.auth.adapters.repositories.async_user_repository import AsyncUserRepository


class AsyncCustomerRepository(AsyncRepository[CustomerDB, int]):
    def __init__(self, session: AsyncSession) -> None:
        self._session = session
        self.user_repository = AsyncUserRepository(session)

    async def create(self, customer: CustomerEntity) -> CustomerEntity:
        user = await self.user_repository.get_by_id(customer.user.id)
        if not user:
            raise UserNotFound
        customer.user = user
        customer_model = CustomerMapper.entity_to_model(customer)
        self.session.add(customer_model)
        await self.session.flush()
        return CustomerMapper.model_to_entity(customer_model, user)

    async def get_by_id(self, id: int) -> CustomerEntity:
        customer_model = await self.find_by_col(user=id)
        user = await self.user_repository.get_by_id(id)
        return CustomerMapper.model_to_entity(customer_model, user)

    async def get_all_by_pagination(
        self, page: int, per_page: int
    ) -> tuple[List[CustomerEntity], int]:
        customers_model, count = await self.paginate(page=page, per_page=per_page)
        customers = []
        for customer in customers_model:
            user = await self.user_repository.get_by_id(customer.user)
            customer = CustomerMapper.model_to_entity(customer, user)
            customers.append(customer)
        return customers, count

    async def delete_by_id(self, id: int):
        customer = await self.find_by_col(user=id)
        if not customer:
            raise CustomerNotFound
        await self.delete(customer)

    async def update(self, id: int, customer_data: dict) -> CustomerEntity:
        customer = await self.find_by_col(user=id)
        user = await self.user_repository.get_by_id(id=id)
        if not customer:
            raise CustomerNotFound
        user_dict = customer_data["user"]
        if user_dict:
            # TODO add user update
            print("update user")
            customer_data.pop("user")

        for key, value in customer_data.items():
            setattr(customer, key, value)
        await self.session.flush()
        return CustomerMapper.model_to_entity(customer, user)

class CustomerRegister(BaseUseCase[CustomerUnitOfWork]):
    def __init__(self, uow: CustomerUnitOfWork) -> None:
        self._uow = uow
        self.user_register = UserRegister(UserUnitOfWork(uow.engine))

    @async_transactional(propagation=Propagation.REQUIRED)
    async def register(self, command: RegisterCustomerCommand) -> Customer:
        user = await self.user_register.execute(command.user)
        customer = Customer(user)
        try:
            customer = await self._uow.repository.create(customer)
            return customer
        except Exception as e:
            raise e

class AsyncReservationRepository(AsyncRepository[ReservationDB, int]):
    def __init__(self, session: AsyncSession) -> None:
        self._session = session
        self.book_repo = AsyncBookRepository(session)
        self.customer_repo = AsyncCustomerRepository(session)

        
    async def create(self, instance: ReservationEntity) -> ReservationEntity:
        reservation_model = ReservationMapper.entity_to_model(instance)
        self.session.add(reservation_model)
        await self.session.flush()
        return ReservationMapper.model_to_entity(reservation_model)

    async def get_by_id(self, id: int) -> ReservationEntity:
        reservation_model = await self.find_by_col(id=id)
        if not reservation_model:
            raise NotFoundException(detail="Reservation not found")
        return ReservationMapper.model_to_entity(reservation_model)

    async def get_all_by_pagination(
        self, page: int, per_page: int
    ) -> tuple[List[ReservationEntity], int]:
        reservations_model, count = await self.paginate(page=page, per_page=per_page)
        reservations = [
            ReservationMapper.model_to_entity(reservation)
            for reservation in reservations_model
        ]
        return reservations, count

    async def update(self, id: int, reservation_data) -> ReservationEntity:
        reservation_model = await self.find_by_col(id=id)
        if not reservation_model:
            raise NotFoundException(detail="Reservation not found")
        if (
            "book_id" in reservation_data
            and reservation_model.book_id != reservation_data["book_id"]
        ):
            book = await self.book_repo.find_by_col(id=reservation_data["book_id"])
            reservation_model.book_id = book.id
        if ("customer_id" in reservation_data) and (
            reservation_model.customer_id != reservation_data["customer_id"]
        ):
            customer = await self.customer_repo.find_by_col(
                id=reservation_data["customer_id"]
            )
            reservation_model.customer_id = customer.id

        for key, value in reservation_data.items():
            setattr(reservation_model, key, value)

        await self.session.flush()
        return ReservationMapper.model_to_entity(reservation_model)

    async def delete(self, instance: ReservationEntity):
        reservation_model = await self.find_by_col(id=instance.id)
        if not reservation_model:
            raise NotFoundException(detail="Reservation not found")
        reservation_model.is_deleted = True

@dataclass
class Reservation(Entity):
    id: Optional[int] = None
    book_id: int = None
    customer_id: int = None
    start_time: datetime = None
    end_time: Optional[datetime] = None
    is_scheduled: bool = False
    is_instant: bool = False
    price: int = 0

    @staticmethod
    def schedule_reservation(customer_id: int, book_id: int) -> "Reservation":
        reservation = Reservation(customer_id=customer_id, book_id=book_id)
        reservation._schedule_reservation(customer_id, book_id)

    @staticmethod
    def instant_reservation(customer_id: int, book_id: int) -> "Reservation":
        reservation = Reservation(customer_id=customer_id, book_id=book_id)
        reservation._instant_reservation(customer_id, book_id)
        return reservation

    def _schedule_reservation(self, customer_id: int, book_id: int):
        self.customer_id = customer_id
        self.book_id = book_id
        self.is_scheduled = True

    def _instant_reservation(self, customer_id: int, book_id: int):
        self.customer_id = customer_id
        self.book_id = book_id
        self.is_instant = True
        self.start_time = datetime.now()

class CreateReservationHandler(BaseUseCase[ReservationUnitOfWork]):
    def __init__(self, uow: ReservationUnitOfWork) -> None:
        self._uow = uow

    @async_transactional(propagation=Propagation.REQUIRED)
    async def execute(self, command: CreateReservationCommand) -> Reservation:
        customer = await self._uow.customer_repository.get_by_id(command.customer_id)
        if not customer:
            raise NotFoundException(detail="Customer not found")

        book = await self._uow.book_repository.get_by_id(command.book_id)
        if not book:
            raise NotFoundException(detail="Book not found")

        if command.is_instant:
            if book.is_empty():
                raise InsufficientFundsException(
                    detail="Not enough units available for instant reservation"
                )
            if customer.wallet_money < command.price:
                raise InsufficientFundsException(detail="Insufficient funds")

            book.decrease_stock_by_one()
            customer.pay_money(command.price)
            reservation = Reservation.instant_reservation(
                customer_id=customer.user.id, book_id=book.id
            )
        else:
            reservation = Reservation.schedule_reservation(
                customer_id=customer.user.id, book_id=book.id
            )

        reservation = await self._uow.repository.create(reservation)
        book = await self._uow.book_repository.update(command.book_id, book.dict)
        customer = await self._uow.customer_repository.update(
            command.customer_id, customer.dict
        )
        await self._uow.commit()
        return reservation

class UpdateReservationHandler(BaseUseCase[ReservationUnitOfWork]):

        if not reservation:
            raise NotFoundException(detail="Reservation not found")

        if command.customer_id:
            customer = await self._uow.repository.customer_repo.get_by_id(

                raise InsufficientFundsException(detail="Insufficient funds")

        await self._uow.reservation_repository.update(reservation)
        await self._uow.commit()
        return reservation

from abc import ABC, abstractmethod
from typing import Any, Generic, TypeVar

from src.shared_kernel.domain.abstract_entity import Entity


MapperEntity = TypeVar("MapperEntity", bound=Entity)
MapperModel = TypeVar("MapperModel", bound=Any)


class DataMapper(Generic[MapperEntity, MapperModel], ABC):
    """
    A generic base class for mapping between domain entities and data models.

    Provides abstract methods to convert data models to domain entities and vice versa.
    Subclasses must implement the conversion logic in `model_to_entity` and `entity_to_model`.

    Usage:
        - Implement `model_to_entity` to convert data models to entities.
        - Implement `entity_to_model` to convert entities to data models.
    """

    entity_class: type[MapperEntity]
    model_class: type[MapperModel]

    @staticmethod
    def model_to_entity(instance: MapperModel) -> MapperEntity:
        return MapperEntity(**instance.dict)

    @staticmethod
    def entity_to_model(entity: MapperEntity) -> MapperModel:
        return MapperModel(**entity.dict)

    """Converts CamelCase to snake_case for table names."""

    This configures the context with just a URL

from sqlalchemy import func
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy.future import select
from typing import (
    TypeVar,
    Tuple,
    List,
    Optional,
    final,
    get_args,
    List,
    Protocol,
    Optional,
    TypeVar,
)
from sqlalchemy.inspection import inspect
from sqlalchemy.sql.selectable import Select

from src.shared_kernel.database.async_postgres_database import CustomBase


_MT = TypeVar("_MT", bound=CustomBase)  # Model Type
_T = TypeVar("_T")  # Primary key Type


class BaseAsyncRepository(Protocol):
    _session: AsyncSession

    @property
    def session(self) -> AsyncSession:
        assert self._session is not None
        return self._session


class AsyncRepository(BaseAsyncRepository, Protocol[_MT, _T]):
    @property
    def _model(self):
        return get_args(self.__orig_bases__[0])[0]

    @property
    def _pk_column(self) -> str:
        return inspect(self._model).primary_key[0].name

    async def delete(self, item: _MT):
        await self.session.delete(item)

    async def find_by_pk(self, pk: _T) -> Optional[_MT]:
        return await self.find_by_col(**{self._pk_column: pk})

    @final
    async def find_by_col(self, **kwargs) -> Optional[_MT]:
        item = await self.session.execute(self._gen_stmt_for_param(**kwargs))
        return item.unique().scalars().one_or_none()

    @final
    def _gen_stmt_for_param(self, **kwargs) -> Select:
        stmt = select(self._model)
        if kwargs:
            for key, value in kwargs.items():
                stmt = stmt.where(getattr(self._model, key) == value)
        return stmt

    @final
    async def find_all(self, **kwargs) -> List[_MT]:
        stmt = self._gen_stmt_for_param(**kwargs)
        result = await self.session.execute(stmt)

        return result.unique().scalars().fetchall()

    @final
    async def is_exists(self, **kwargs) -> bool:
        result = await self.session.execute(
            self._gen_stmt_for_param(**kwargs).exists().select()
        )
        return result.scalar()

    @final
    def create(self, item: _MT):
        self.session.add(item)

    @final
    async def create_all(self, items: List[_MT]):
        self.session.add_all(items)

    def update(self, item: _MT, req: dict):
        for k, v in req.items():
            if v is not None:
                setattr(item, k, v)

    @final
    async def paginate(
        self, page: int, per_page: int, **kwargs
    ) -> Tuple[List[_MT], int]:
        stmt = self._gen_stmt_for_param(**kwargs)

        offset = max((page - 1) * per_page, 0)
        paginated_query = stmt.offset(offset).limit(per_page)

        result = await self.session.execute(paginated_query)
        items = result.unique().scalars().all()

        total_count = await self.session.execute(
            select(func.count()).select_from(self._model)
        )
        total_count = total_count.scalar()

        return items, total_count

import schedule
import time
from threading import Thread

class Scheduler:
    def __init__(self):
        self.jobs = []

    def every(self, interval):
        job = schedule.every(interval)
        self.jobs.append(job)
        return job

    def run_pending(self):
        while True:
            schedule.run_pending()
            time.sleep(1)

    def start(self):
        thread = Thread(target=self.run_pending)
        thread.daemon = True
        thread.start()

class BaseUseCase(Protocol[_UT]):

from abc import ABC, abstractmethod
from sqlalchemy.ext.asyncio import AsyncEngine, AsyncSession
from typing import Optional, Type


class AsyncBaseUnitOfWork(ABC):
    @abstractmethod
    async def __aenter__(self) -> None:
        raise NotImplementedError("required __aenter__ call for UoW Pattern")

    async def __aexit__(
        self,
        exc_type: Optional[Type[Exception]],
        exc_val: Optional[Exception],
        traceback,
    ):
        if exc_type:
            await self.rollback()

    @abstractmethod
    async def commit(self):
        raise NotImplementedError("Choice ORM commit func")

    @abstractmethod
    async def rollback(self):
        raise NotImplementedError("Choice ORM rollback func")


class AsyncSQLAlchemyUnitOfWork(AsyncBaseUnitOfWork):
    def __init__(self, engine: AsyncEngine) -> None:
        self._engine = engine
        self._session: Optional[AsyncSession] = None

    @property
    def engine(self) -> AsyncEngine:
        assert self._engine is not None
        return self._engine

    @property
    def session(self) -> AsyncSession:
        assert self._session is not None
        return self._session

    async def __aenter__(self):
        self._session = AsyncSession(self.engine)

    async def __aexit__(
        self,
        exc_type: Optional[Type[Exception]],
        exc_val: Optional[Exception],
        traceback,
    ):
        await super().__aexit__(exc_type, exc_val, traceback)
        await self.session.close()

    async def commit(self):
        await self.session.commit()

    async def flush(self):
        await self.session.flush()

    async def refresh(self, item):
        await self.session.refresh(item)

    async def rollback(self):
        await self.session.rollback()

from sqlalchemy import inspect
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy.exc import NoInspectionAvailable
from enum import Enum


class Propagation(Enum):
    REQUIRED = "required"
    REQUIRES_NEW = "requires_new"


async def __async_propagation_required(
    self, func, read_only: bool, session: AsyncSession, args, kwargs
):
    if not session.is_active:
        session.begin(subtransactions=True)

    result = await func(self, *args, **kwargs)
    if not read_only:
        await session.commit()
        try:
            inspect(result)
            await session.refresh(result)
        except NoInspectionAvailable:
            pass

    return result


async def __async_requires_new(
    self, func, read_only: bool, session: AsyncSession, args, kwargs
):
    if not session.is_active:
        session.begin()

    result = await func(self, *args, **kwargs)
    if not read_only:
        await session.commit()
        try:
            inspect(result)
            await session.refresh(result)
        except NoInspectionAvailable:
            pass

    return result


def async_transactional(
    read_only: bool = False, propagation: Propagation = Propagation.REQUIRES_NEW
):
    def decorator(func):
        async def wrapper(self, *args, **kwargs):
            if hasattr(self, "uow"):
                async with self.uow:
                    if propagation == Propagation.REQUIRED:
                        result = await __async_propagation_required(
                            self=self,
                            func=func,
                            read_only=read_only,
                            session=self.uow.session,
                            args=args,
                            kwargs=kwargs,
                        )
                    else:
                        result = await __async_requires_new(
                            self=self,
                            func=func,
                            read_only=read_only,
                            session=self.uow.session,
                            args=args,
                            kwargs=kwargs,
                        )

                    return result

            elif hasattr(self, "session"):
                if propagation == Propagation.REQUIRED:
                    result = await __async_propagation_required(
                        self=self,
                        func=func,
                        read_only=read_only,
                        session=self.session,
                        args=args,
                        kwargs=kwargs,
                    )
                else:
                    result = await __async_requires_new(
                        self=self,
                        func=func,
                        read_only=read_only,
                        session=self.session,
                        args=args,
                        kwargs=kwargs,
                    )

                return result

        return wrapper

    return decorator

===== wait-for-it.sh =====
#!/usr/bin/env bash
# Use this script to test if a given TCP host/port are available

WAITFORIT_cmdname=${0##*/}

echoerr() { if [[ $WAITFORIT_QUIET -ne 1 ]]; then echo "$@" 1>&2; fi }

usage()
{
    cat << USAGE >&2
Usage:
    $WAITFORIT_cmdname host:port [-s] [-t timeout] [-- command args]
    -h HOST | --host=HOST       Host or IP under test
    -p PORT | --port=PORT       TCP port under test
                                Alternatively, you specify the host and port as host:port
    -s | --strict               Only execute subcommand if the test succeeds
    -q | --quiet                Don't output any status messages
    -t TIMEOUT | --timeout=TIMEOUT
                                Timeout in seconds, zero for no timeout
    -- COMMAND ARGS             Execute command with args after the test finishes
USAGE
    exit 1
}

wait_for()
{
    if [[ $WAITFORIT_TIMEOUT -gt 0 ]]; then
        echoerr "$WAITFORIT_cmdname: waiting $WAITFORIT_TIMEOUT seconds for $WAITFORIT_HOST:$WAITFORIT_PORT"
    else
        echoerr "$WAITFORIT_cmdname: waiting for $WAITFORIT_HOST:$WAITFORIT_PORT without a timeout"
    fi
    WAITFORIT_start_ts=$(date +%s)
    while :
    do
        if [[ $WAITFORIT_ISBUSY -eq 1 ]]; then
            nc -z $WAITFORIT_HOST $WAITFORIT_PORT
            WAITFORIT_result=$?
        else
            (echo -n > /dev/tcp/$WAITFORIT_HOST/$WAITFORIT_PORT) >/dev/null 2>&1
            WAITFORIT_result=$?
        fi
        if [[ $WAITFORIT_result -eq 0 ]]; then
            WAITFORIT_end_ts=$(date +%s)
            echoerr "$WAITFORIT_cmdname: $WAITFORIT_HOST:$WAITFORIT_PORT is available after $((WAITFORIT_end_ts - WAITFORIT_start_ts)) seconds"
            break
        fi
        sleep 1
    done
    return $WAITFORIT_result
}

wait_for_wrapper()
{
    # In order to support SIGINT during timeout: http://unix.stackexchange.com/a/57692
    if [[ $WAITFORIT_QUIET -eq 1 ]]; then
        timeout $WAITFORIT_BUSYTIMEFLAG $WAITFORIT_TIMEOUT $0 --quiet --child --host=$WAITFORIT_HOST --port=$WAITFORIT_PORT --timeout=$WAITFORIT_TIMEOUT &
    else
        timeout $WAITFORIT_BUSYTIMEFLAG $WAITFORIT_TIMEOUT $0 --child --host=$WAITFORIT_HOST --port=$WAITFORIT_PORT --timeout=$WAITFORIT_TIMEOUT &
    fi
    WAITFORIT_PID=$!
    trap "kill -INT -$WAITFORIT_PID" INT
    wait $WAITFORIT_PID
    WAITFORIT_RESULT=$?
    if [[ $WAITFORIT_RESULT -ne 0 ]]; then
        echoerr "$WAITFORIT_cmdname: timeout occurred after waiting $WAITFORIT_TIMEOUT seconds for $WAITFORIT_HOST:$WAITFORIT_PORT"
    fi
    return $WAITFORIT_RESULT
}

# process arguments
while [[ $# -gt 0 ]]
do
    case "$1" in
        *:* )
        WAITFORIT_hostport=(${1//:/ })
        WAITFORIT_HOST=${WAITFORIT_hostport[0]}
        WAITFORIT_PORT=${WAITFORIT_hostport[1]}
        shift 1
        ;;
        --child)
        WAITFORIT_CHILD=1
        shift 1
        ;;
        -q | --quiet)
        WAITFORIT_QUIET=1
        shift 1
        ;;
        -s | --strict)
        WAITFORIT_STRICT=1
        shift 1
        ;;
        -h)
        WAITFORIT_HOST="$2"
        if [[ $WAITFORIT_HOST == "" ]]; then break; fi
        shift 2
        ;;
        --host=*)
        WAITFORIT_HOST="${1#*=}"
        shift 1
        ;;
        -p)
        WAITFORIT_PORT="$2"
        if [[ $WAITFORIT_PORT == "" ]]; then break; fi
        shift 2
        ;;
        --port=*)
        WAITFORIT_PORT="${1#*=}"
        shift 1
        ;;
        -t)
        WAITFORIT_TIMEOUT="$2"
        if [[ $WAITFORIT_TIMEOUT == "" ]]; then break; fi
        shift 2
        ;;
        --timeout=*)
        WAITFORIT_TIMEOUT="${1#*=}"
        shift 1
        ;;
        --)
        shift
        WAITFORIT_CLI=("$@")
        break
        ;;
        --help)
        usage
        ;;
        *)
        echoerr "Unknown argument: $1"
        usage
        ;;
    esac
done

if [[ "$WAITFORIT_HOST" == "" || "$WAITFORIT_PORT" == "" ]]; then
    echoerr "Error: you need to provide a host and port to test."
    usage
fi

WAITFORIT_TIMEOUT=${WAITFORIT_TIMEOUT:-15}
WAITFORIT_STRICT=${WAITFORIT_STRICT:-0}
WAITFORIT_CHILD=${WAITFORIT_CHILD:-0}
WAITFORIT_QUIET=${WAITFORIT_QUIET:-0}

# Check to see if timeout is from busybox?
WAITFORIT_TIMEOUT_PATH=$(type -p timeout)
WAITFORIT_TIMEOUT_PATH=$(realpath $WAITFORIT_TIMEOUT_PATH 2>/dev/null || readlink -f $WAITFORIT_TIMEOUT_PATH)

WAITFORIT_BUSYTIMEFLAG=""
if [[ $WAITFORIT_TIMEOUT_PATH =~ "busybox" ]]; then
    WAITFORIT_ISBUSY=1
    # Check if busybox timeout uses -t flag
    # (recent Alpine versions don't support -t anymore)
    if timeout &>/dev/stdout | grep -q -e '-t '; then
        WAITFORIT_BUSYTIMEFLAG="-t"
    fi
else
    WAITFORIT_ISBUSY=0
fi

if [[ $WAITFORIT_CHILD -gt 0 ]]; then
    wait_for
    WAITFORIT_RESULT=$?
    exit $WAITFORIT_RESULT
else
    if [[ $WAITFORIT_TIMEOUT -gt 0 ]]; then
        wait_for_wrapper
        WAITFORIT_RESULT=$?
    else
        wait_for
        WAITFORIT_RESULT=$?
    fi
fi

if [[ $WAITFORIT_CLI != "" ]]; then
    if [[ $WAITFORIT_RESULT -ne 0 && $WAITFORIT_STRICT -eq 1 ]]; then
        echoerr "$WAITFORIT_cmdname: strict mode, refusing to execute subprocess"
        exit $WAITFORIT_RESULT
    fi
    exec "${WAITFORIT_CLI[@]}"
else
    exit $WAITFORIT_RESULT
fi
